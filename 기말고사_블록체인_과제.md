import asyncio
import json

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client


def print_json(data):
    print(json.dumps(data, ensure_ascii=False, indent=2))


async def main():
    server_params = StdioServerParameters(
        command="python",
        args=["server.py"]
    )

    async with stdio_client(server_params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()

            while True:
                print("\n========== Upbit MCP Client ==========")
                print("1. Get Market List")
                print("2. Get Current Ticker")
                print("3. Get Orderbook")
                print("4. Get Recent Trades")
                print("5. Get Daily Candles")
                print("0. Exit")

                choice = input("Select Menu: ")

                if choice == "0":
                    print("Program terminated.")
                    break

                elif choice == "1":
                    result = await session.call_tool(
                        "get_markets",
                        {}
                    )

                    print_json(result.content[0].text)

                elif choice == "2":
                    market = input("Enter Market Example KRW-BTC: ")

                    result = await session.call_tool(
                        "get_ticker",
                        {
                            "market": market
                        }
                    )

                    print_json(result.content[0].text)

                elif choice == "3":
                    market = input("Enter Market Example KRW-BTC: ")

                    result = await session.call_tool(
                        "get_orderbook",
                        {
                            "market": market
                        }
                    )

                    print_json(result.content[0].text)

                elif choice == "4":
                    market = input("Enter Market Example KRW-BTC: ")
                    count = int(input("Enter Count Example 10: "))

                    result = await session.call_tool(
                        "get_recent_trades",
                        {
                            "market": market,
                            "count": count
                        }
                    )

                    print_json(result.content[0].text)

                elif choice == "5":
                    market = input("Enter Market Example KRW-BTC: ")
                    count = int(input("Enter Count Example 5: "))

                    result = await session.call_tool(
                        "get_day_candles",
                        {
                            "market": market,
                            "count": count
                        }
                    )

                    print_json(result.content[0].text)

                else:
                    print("Invalid selection.")


if __name__ == "__main__":
    asyncio.run(main())


server.py

from mcp.server.fastmcp import FastMCP
import requests

mcp = FastMCP("Upbit MCP Server")

BASE_URL = "https://api.upbit.com"


def upbit_get(path, params=None):
    url = BASE_URL + path

    headers = {
        "Accept": "application/json"
    }

    response = requests.get(
        url,
        headers=headers,
        params=params,
        timeout=10
    )

    if response.status_code != 200:
        return {
            "error": True,
            "status_code": response.status_code,
            "message": response.text
        }

    return response.json()


@mcp.tool()
def get_markets():
    """
    Get all available markets from Upbit.
    Example: KRW-BTC, KRW-ETH
    """

    return upbit_get(
        "/v1/market/all",
        {
            "isDetails": "false"
        }
    )


@mcp.tool()
def get_ticker(market: str = "KRW-BTC"):
    """
    Get current ticker information.
    """

    return upbit_get(
        "/v1/ticker",
        {
            "markets": market
        }
    )


@mcp.tool()
def get_orderbook(market: str = "KRW-BTC"):
    """
    Get orderbook information.
    """

    return upbit_get(
        "/v1/orderbook",
        {
            "markets": market
        }
    )


@mcp.tool()
def get_recent_trades(
    market: str = "KRW-BTC",
    count: int = 10
):
    """
    Get recent trade history.
    """

    return upbit_get(
        "/v1/trades/ticks",
        {
            "market": market,
            "count": count
        }
    )


@mcp.tool()
def get_day_candles(
    market: str = "KRW-BTC",
    count: int = 5
):
    """
    Get daily candle data.
    """

    return upbit_get(
        "/v1/candles/days",
        {
            "market": market,
            "count": count
        }
    )


if __name__ == "__main__":
    mcp.run()
